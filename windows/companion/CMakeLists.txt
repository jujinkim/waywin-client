add_executable(waywin-companion WIN32 main.cpp ../probe/probe.rc)
target_include_directories(waywin-companion PRIVATE ../../protocol ../probe)
target_compile_features(waywin-companion PRIVATE cxx_std_17)
target_compile_definitions(waywin-companion PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(waywin-companion PRIVATE user32 gdi32 shell32 ole32 oleaut32 uuid wtsapi32 windowscodecs advapi32 dwmapi)
if(MSVC)
  target_compile_options(waywin-companion PRIVATE /W4 /WX /utf-8)
  target_link_options(waywin-companion PRIVATE /MANIFEST:NO)
else()
  target_compile_options(waywin-companion PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MINGW)
  target_link_options(waywin-companion PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
install(TARGETS waywin-companion RUNTIME DESTINATION bin)
