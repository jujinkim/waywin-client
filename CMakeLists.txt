cmake_minimum_required(VERSION 3.22)
project(WayWin VERSION 0.3.0 LANGUAGES CXX)

include(CTest)
option(WAYWIN_BUILD_APP "Build the Qt desktop product" ON)
option(WAYWIN_BUILD_RUNTIME "Build the pinned SDL session runtime" OFF)
if(WIN32)
  set(WAYWIN_BUILD_APP OFF)
endif()
if(WAYWIN_BUILD_RUNTIME)
  include(cmake/ProductRuntime.cmake)
endif()
if(WAYWIN_BUILD_APP)
  add_subdirectory(app)
endif()

if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME probe-contracts
    COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -v)
  set_tests_properties(probe-contracts PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
endif()
if(WIN32)
  enable_language(RC)
  add_subdirectory(windows/probe)
  add_subdirectory(windows/companion)
endif()
include(cmake/ProductPackaging.cmake)
